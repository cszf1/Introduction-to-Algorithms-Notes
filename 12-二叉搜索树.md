| 数据结构 | 查找复杂度 | 插入复杂度 | 删除复杂度 | 有序遍历 |
|:--:|:--:|:--:|:--:|:--:|
| 数组（无序） | O(n) | O(1) | O(n) | ❌ |
| 数组（有序） | O(log n) | O(n) | O(n) | ✅ O(n) |

**核心优势**：同时支持高效查找、插入、删除，且天然支持有序遍历！

## 12.1 二叉搜索树的定义

### 📖 什么是二叉搜索树？

**二叉搜索树（Binary Search Tree, BST）** 是一种特殊的二叉树，其中每个节点包含一个键值，并且满足以下性质：

```
┌─────────────────────────────────────────────────────────┐
│  二叉搜索树性质（BST Property）：                         │
│                                                         │
│  对于任意节点 x：                                        │
│    - x.left（小）：x的所有左子树节点 ≤ x.key            │
│    - x.right（大）：x的所有右子树节点 ≥ x.key           │
│                                                         │
│  递归定义：                                               │
│    - 左子树所有节点 < 当前节点                           │
│    - 右子树所有节点 > 当前节点                           │
│    - 左右子树本身也是二叉搜索树                          │
└─────────────────────────────────────────────────────────┘
```

#### Python 实现

```python
# 二叉搜索树节点的 Python 定义
class TreeNode:
    def __init__(self, key, left=None, right=None, parent=None):
        """
        初始化一个二叉搜索树节点
        
        参数：
            key: 节点的值（键）
            left: 左子节点（默认为 None）
            right: 右子节点（默认为 None）
            parent: 父节点（默认为 None）
        """
        self.key = key       # 节点存储的关键字
        self.left = left     # 指向左子节点的指针
        self.right = right   # 指向右子节点的指针
        self.parent = parent # 指向父节点的指针（方便某些操作）


class BinarySearchTree:
    def __init__(self):
        """初始化一个空二叉搜索树"""
        self.root = None  # 树的根节点，初始为空
    
    def insert(self, key):
        """插入一个新节点"""
        # 创建新节点
        new_node = TreeNode(key)
        
        # 如果树为空，新节点就是根节点
        if self.root is None:
            self.root = new_node
            return
        
        # 否则，从根节点开始寻找插入位置
        current = self.root  # 当前遍历的节点
        while True:
            if key < current.key:
                # 如果新key小于当前节点，去左边
                if current.left is None:
                    # 左边为空，直接插入
                    current.left = new_node
                    new_node.parent = current
                    break
                current = current.left  # 继续向左下探
            else:
                # 如果新key大于等于当前节点，去右边
                if current.right is None:
                    # 右边为空，直接插入
                    current.right = new_node
                    new_node.parent = current
                    break
                current = current.right  # 继续向右下探
```

#### C++ 实现

```cpp
// 二叉搜索树节点的 C++ 定义
struct TreeNode {
    int key;                // 节点存储的关键字
    TreeNode* left;         // 指向左子节点的指针
    TreeNode* right;        // 指向右子节点的指针
    TreeNode* parent;       // 指向父节点的指针
    
    // 构造函数：初始化节点
    TreeNode(int k, TreeNode* l = nullptr, TreeNode* r = nullptr, TreeNode* p = nullptr)
        : key(k), left(l), right(r), parent(p) {}
};


class BinarySearchTree {
private:
    TreeNode* root;  // 树的根节点
    
public:
    // 构造函数：初始化空树
    BinarySearchTree() : root(nullptr) {}
    
    // 插入一个新节点
    void insert(int key) {
        // 创建新节点
        TreeNode* new_node = new TreeNode(key);
        
        // 如果树为空，新节点就是根节点
        if (root == nullptr) {
            root = new_node;
            return;
        }
        
        // 从根节点开始寻找插入位置
        TreeNode* current = root;
        while (true) {
            if (key < current->key) {
                // 新key小于当前节点，去左边
                if (current->left == nullptr) {
                    current->left = new_node;      // 插入作为左子节点
                    new_node->parent = current;    // 设置父节点指针
                    break;
                }
                current = current->left;           // 继续向左下探
            } else {
                // 新key大于等于当前节点，去右边
                if (current->right == nullptr) {
                    current->right = new_node;     // 插入作为右子节点
                    new_node->parent = current;    // 设置父节点指针
                    break;
                }
                current = current->right;          // 继续向右下探
            }
        }
    }
};
```

## 12.2 查询操作

### 📖 二叉搜索树的核心查询操作

| 操作 | 功能 | 时间复杂度 |

#### 📝 代码实现

##### Python 实现（递归版本）

```python
def search(self, key):
    """
    在二叉搜索树中查找关键字 key
    
    参数：
        key: 要查找的关键字
    
    返回：
        找到的节点对象，如果不存在则返回 None
    
    时间复杂度：O(h)，h 为树的高度
    """
    # 从根节点开始递归查找
    return self._search_recursive(self.root, key)


def _search_recursive(node, key):
    """
    递归查找的辅助函数
    
    参数：
        node: 当前遍历的节点
        key: 要查找的关键字
    
    返回：
        找到的节点或 None
    """
    # 基本情况1：节点为空（没找到）
    if node is None:
        return None
    
    # 基本情况2：找到了！
    if key == node.key:
        return node
    
    # 递归情况：根据大小关系选择子树
    if key < node.key:
        # key 比当前节点小，去左子树找
        return self._search_recursive(node.left, key)
    else:
        # key 比当前节点大（或相等），去右子树找
        return self._search_recursive(node.right, key)
```

##### Python 实现（迭代版本 - 更高效）

```python
def search_iterative(self, key):
    """
    迭代版本的查找（推荐使用，避免递归栈溢出）
    
    参数：
        key: 要查找的关键字
    
    返回：
        找到的节点对象，如果不存在则返回 None
    """
    current = self.root  # 从根节点开始
    
    # 只要当前节点不为空，就继续查找
    while current is not None:
        if key == current.key:
            # 找到了！
            return current
        elif key < current.key:
            # key 更小，去左边
            current = current.left
        else:
            # key 更大（或相等），去右边
            current = current.right
    
    # 遍历完都没找到
    return None
```

##### C++ 实现

```cpp
// 递归版本
TreeNode* searchRecursive(TreeNode* node, int key) {
    // 基本情况1：节点为空
    if (node == nullptr) {
        return nullptr;
    }
    
    // 基本情况2：找到了
    if (key == node->key) {
        return node;
    }
    
    // 递归情况：根据大小选择子树
    if (key < node->key) {
        return searchRecursive(node->left, key);   // 去左子树找
    } else {
        return searchRecursive(node->right, key);   // 去右子树找
    }
}

// 包装函数
TreeNode* search(int key) {
    return searchRecursive(root, key);
}

// 迭代版本（推荐）
TreeNode* searchIterative(int key) {
    TreeNode* current = root;  // 从根节点开始
    
    while (current != nullptr) {
        if (key == current->key) {
            return current;     // 找到了
        } else if (key < current->key) {
            current = current->left;   // 去左子树
        } else {
            current = current->right;   // 去右子树
        }
    }
    
    return nullptr;  // 没找到
}
```

#### 📝 代码实现

##### Python 实现

```python
def minimum(self):
    """
    查找树中的最小值节点
    
    返回：最小值节点，如果树为空则返回 None
    """
    if self.root is None:
        return None
    
    current = self.root
    # 一直往左走，直到左子节点为空
    while current.left is not None:
        current = current.left
    
    return current


def maximum(self):
    """
    查找树中的最大值节点
    
    返回：最大值节点，如果树为空则返回 None
    """
    if self.root is None:
        return None
    
    current = self.root
    # 一直往右走，直到右子节点为空
    while current.right is not None:
        current = current.right
    
    return current


# 递归版本（理解用）
def minimum_recursive(self, node):
    """递归找最小值"""
    if node is None:
        return None
    
    if node.left is None:
        return node  # 没有左子树，当前节点就是最小的
    
    return self.minimum_recursive(node.left)  # 去左子树找
```

##### C++ 实现

```cpp
    }

}
```

### 12.2.3 前驱和后继（SUCCESSOR & PREDECESSOR）

#### 📖 什么是前驱和后继？

```
┌─────────────────────────────────────────────────────────┐
│  前驱（Predecessor）：                                   │
│  - 在中序遍历中，位于给定节点"前面"的节点               │
│  - 即比给定节点小的"最近"的那个节点                      │
│                                                         │
│  后继（Successor）：                                     │
│  - 在中序遍历中，位于给定节点"后面"的节点                │
│  - 即比给定节点大的"最近"的那个节点                      │
└─────────────────────────────────────────────────────────┘
```

#### 📊 过程图解：后继查找

```
中序遍历序列：1 → 3 → 4 → 6 → 7 → 8 → 10 → 13 → 14

┌────────────────────────────────────────────────────────┐
│  找 6 的后继：                                          │
│                                                        │
│                          8 ← 后继在这里！               │
│                        /    \                          │
│                       /      \                         │
│                      3       10                        │
│                    /   \       \                       │
│                   1     6 ← 目标   14                  │
│                        /   \                          │
│                       4     7                          │
│                                                        │
│  分析：                                                 │
│  - 6 的右子树存在：{4, 7}                               │
```

```python
def successor(self, node):
    """
    查找给定节点的后继
    
    参数：
        node: 给定的节点
    
    返回：
        后继节点，如果不存在则返回 None
    """
    # Case 1：右子树存在
    if node.right is not None:
        # 后继 = 右子树中的最小值
        return self.minimum_recursive(node.right)
    
    # Case 2 & 3：右子树不存在
    # 向上找第一个作为左子节点的祖先
    current = node
    parent = node.parent
    
    while parent is not None and current == parent.right:
        # current 是 parent 的右子节点，继续往上找
        current = parent
        parent = parent.parent
    
    return parent  # 后继或 None（自己是最大值）


def predecessor(self, node):
    """
    查找给定节点的前驱（与后继对称）
    
    参数：
        node: 给定的节点
    
    返回：
        前驱节点，如果不存在则返回 None
    """
    # Case 1：左子树存在
    if node.left is not None:
        # 前驱 = 左子树中的最大值
        return self.maximum_in_subtree(node.left)
    
    # Case 2 & 3：左子树不存在
    # 向上找第一个作为右子节点的祖先
    current = node
    parent = node.parent
    
    while parent is not None and current == parent.left:
        # current 是 parent 的左子节点，继续往上找
        current = parent
        parent = parent.parent
```

```python
def insert(self, key):
    """
    向二叉搜索树中插入一个新节点
    
    参数：
        key: 要插入的关键字
    
    时间复杂度：O(h)
    """
    # Step 1：创建新节点
    new_node = TreeNode(key)
    
    # Step 2：如果是空树
    if self.root is None:
        self.root = new_node
        return
    
    # Step 3：从根节点开始找插入位置
    parent = None      # 记录父节点
    current = self.root  # 当前遍历的节点
    
    while current is not None:
        parent = current  # 先更新父节点
        
        if key < current.key:
            # 新key更小，去左子树
            if current.left is None:
                # 左子树为空，插入到这里
                current.left = new_node
                new_node.parent = parent
                return
            current = current.left  # 继续下探
        else:
            # 新key更大（或相等），去右子树
            if current.right is None:
                # 右子树为空，插入到这里
                current.right = new_node
                new_node.parent = parent
                return
            current = current.right  # 继续下探


# 递归版本（理解用）
def insert_recursive(self, key):
    """递归版本的插入"""
    self.root = self._insert_recursive(self.root, key, None)


def _insert_recursive(self, node, key, parent):
    """
    递归插入的辅助函数
    
    参数：
        node: 当前遍历的节点
        key: 要插入的关键字
        parent: 当前节点的父节点
    
    返回：
        插入后的子树根节点
    """
    # 基本情况：找到空位
    if node is None:
        new_node = TreeNode(key)
        new_node.parent = parent
        return new_node
    
    # 递归情况
    if key < node.key:
        node.left = self._insert_recursive(node.left, key, node)
    else:
        node.right = self._insert_recursive(node.right, key, node)
    
    return node
```

```cpp
// 插入操作
void insert(int key) {
    // Step 1：创建新节点
    TreeNode* new_node = new TreeNode(key);
    
    // Step 2：如果是空树
    if (root == nullptr) {
        root = new_node;
        return;
    }
    
    // Step 3：从根节点开始找插入位置
    TreeNode* parent = nullptr;
    TreeNode* current = root;
    
    while (current != nullptr) {
        parent = current;
        
        if (key < current->key) {
            // 去左子树
            if (current->left == nullptr) {
                current->left = new_node;
                new_node->parent = parent;
                return;
            }
            current = current->left;
        } else {
            // 去右子树
            if (current->right == nullptr) {
                current->right = new_node;
                new_node->parent = parent;
                return;
            }
            current = current->right;
        }
    }
}
```

### 12.3.2 删除（DELETE）

#### 📖 删除操作的三种情况

删除是二叉搜索树中最复杂的操作，需要考虑三种情况：

```
┌─────────────────────────────────────────────────────────┐
│  删除节点 z 的三种情况：                                 │
```

```python
def delete(self, key):
    """
    从二叉搜索树中删除关键字为 key 的节点
    
    参数：
        key: 要删除的关键字
    
    时间复杂度：O(h)
    """
    # Step 1：找到要删除的节点
    node = self.search_iterative(key)
    if node is None:
        return  # 没找到，不删除任何东西
    
    # Step 2：根据情况删除
    self._delete_node(node)


def _delete_node(self, node):
    """
    删除给定节点的内部方法
    """
    # 如果 node 有两个子节点，转换为找一个子节点的情况
    if node.left is not None and node.right is not None:
        # 找后继（后继一定没有左子树）
        successor = self._minimum_node(node.right)
        # 用后继的值替换当前节点的值
        node.key = successor.key
        # 现在问题转化为删除后继节点
        node = successor
        # 后继一定只有右子树（或没有子节点）
    
    # 现在 node 要么没有子节点，要么只有一个子节点
    # 找到它的唯一子节点（可能是 None）
    child = node.left if node.left is not None else node.right
    
    if child is not None:
        # Case 2：node 有一个子节点
        # 用 child 替代 node
        child.parent = node.parent
        
        if node.parent is None:
            # node 是根节点
            self.root = child
        elif node == node.parent.left:
            # node 是父节点的左子节点
            node.parent.left = child
        else:
            # node 是父节点的右子节点
            node.parent.right = child
    else:
        # Case 1：node 没有子节点（叶子节点）
        if node.parent is None:
            # node 是根节点且是唯一的节点
            self.root = None
        elif node == node.parent.left:
            # node 是父节点的左子节点
            node.parent.left = None
        else:
            # node 是父节点的右子节点
            node.parent.right = None


def _minimum_node(self, node):
    """返回以 node 为根的子树中的最小值节点"""
    current = node
    while current.left is not None:
        current = current.left
    return current
```

```cpp
// 删除操作
void remove(int key) {
    // Step 1：找到要删除的节点
    TreeNode* node = searchIterative(key);
    if (node == nullptr) {
        return;  // 没找到
    }
    
    // Step 2：删除节点
    deleteNode(node);
}

void deleteNode(TreeNode* node) {
    // Case 3：有两个子节点，转换为有一个或没有子节点的情况
    if (node->left != nullptr && node->right != nullptr) {
        // 找后继
        TreeNode* successor = minimumNode(node->right);
        // 用后继的值替换
        node->key = successor->key;
        // 现在删除后继节点
        node = successor;
        // 此时 node 一定只有右子树（或没有子节点）
    }
    
    // 现在 node 要么没有子节点，要么只有一个子节点
    TreeNode* child = (node->left != nullptr) ? node->left : node->right;
    
    if (child != nullptr) {
        // Case 2：有一个子节点
        child->parent = node->parent;
        
        if (node->parent == nullptr) {
            root = child;
        } else if (node == node->parent->left) {
            node->parent->left = child;
        } else {
            node->parent->right = child;
        }
    } else {
        // Case 1：叶子节点
        if (node->parent == nullptr) {
            root = nullptr;
        } else if (node == node->parent->left) {
            node->parent->left = nullptr;
        } else {
            node->parent->right = nullptr;
        }
    }
    
    delete node;  // 释放内存
}

// 辅助函数：找子树最小值
TreeNode* minimumNode(TreeNode* node) {
    TreeNode* current = node;
    while (current->left != nullptr) {
        current = current->left;
    }
    return current;
}
```

## 12.4 随机构建二叉搜索树

### 📖 为什么要讨论随机构建？

```

│  关键问题：树的高度决定操作效率！                         │
│                                                         │
│  最好情况（平衡树）：     最坏情况（退化成链表）：        │

│                                           5               │
│                                                         │
│  平衡树：h ≈ O(log n)                                   │
│  链表：  h = O(n)                                       │
```

### 📊 有序插入 vs 随机插入

```

│  有序插入 1, 2, 3, 4, 5, 6, 7：                         │
│                                                        │
│    1                    退化成链表！                    │
```

```python
class TreeNode:
    """二叉搜索树节点"""
    def __init__(self, key):
        self.key = key           # 节点的关键字
        self.left = None         # 左子节点
        self.right = None        # 右子节点
        self.parent = None       # 父节点


class BinarySearchTree:
    """二叉搜索树完整实现"""
    
    def __init__(self):
        """初始化空树"""
        self.root = None
    
    # ==================== 查找操作 ====================
    
    def search(self, key):
        """
        查找关键字为 key 的节点
        
        时间复杂度：O(h)
        """
        current = self.root
        while current is not None:
            if key == current.key:
                return current
            elif key < current.key:
                current = current.left
            else:
                current = current.right
        return None
    
    def minimum(self):
        """返回树中的最小值节点"""
        if self.root is None:
            return None
        return self._minimum_node(self.root)
    
    def maximum(self):
        """返回树中的最大值节点"""
        if self.root is None:
            return None
        return self._maximum_node(self.root)
    
    def successor(self, node):
        """返回给定节点的后继"""
        if node.right is not None:
            return self._minimum_node(node.right)
        
        parent = node.parent
        while parent is not None and node == parent.right:
            node = parent
            parent = parent.parent
        return parent
    
    def predecessor(self, node):
        """返回给定节点的前驱"""
        if node.left is not None:
            return self._maximum_node(node.left)
        
        parent = node.parent
        while parent is not None and node == parent.left:
            node = parent
            parent = parent.parent
        return parent
    
    # ==================== 修改操作 ====================
    
    def insert(self, key):
        """
        插入关键字为 key 的节点
        
        时间复杂度：O(h)
        """
        new_node = TreeNode(key)
        
        if self.root is None:
            self.root = new_node
            return
        
        parent = None
        current = self.root
        
        while current is not None:
            parent = current
            if key < current.key:
                if current.left is None:
                    current.left = new_node
                    new_node.parent = parent
                    return
                current = current.left
            else:
                if current.right is None:
                    current.right = new_node
                    new_node.parent = parent
                    return
                current = current.right
    
    def delete(self, key):
        """
        删除关键字为 key 的节点
        
        时间复杂度：O(h)
        """
        node = self.search(key)
        if node is None:
            return
        
        self._delete_node(node)
    
    # ==================== 遍历操作 ====================
    
    def inorder_walk(self):
        """
        中序遍历（得到有序序列）
        
        时间复杂度：O(n)
        """
        result = []
        self._inorder_recursive(self.root, result)
        return result
    
    def preorder_walk(self):
        """前序遍历"""
        result = []
        self._preorder_recursive(self.root, result)
        return result
    
    def postorder_walk(self):
        """后序遍历"""
        result = []
        self._postorder_recursive(self.root, result)
        return result
    
    # ==================== 辅助方法 ====================
    
    def _minimum_node(self, node):
        """返回以 node 为根的子树的最小值节点"""
        while node.left is not None:
            node = node.left
        return node
    
    def _maximum_node(self, node):
        """返回以 node 为根的子树的最大值节点"""
        while node.right is not None:
            node = node.right
        return node
    
    def _inorder_recursive(self, node, result):
        """中序遍历递归辅助"""
        if node is not None:
            self._inorder_recursive(node.left, result)
            result.append(node.key)
            self._inorder_recursive(node.right, result)
    
    def _preorder_recursive(self, node, result):
        """前序遍历递归辅助"""
        if node is not None:
            result.append(node.key)
            self._preorder_recursive(node.left, result)
            self._preorder_recursive(node.right, result)
    
    def _postorder_recursive(self, node, result):
        """后序遍历递归辅助"""
        if node is not None:
            self._postorder_recursive(node.left, result)
            self._postorder_recursive(node.right, result)
            result.append(node.key)
    
    def _delete_node(self, node):
        """删除给定节点"""
        # Case 3：两个子节点，转换为只有一个或没有子节点
        if node.left is not None and node.right is not None:
            successor = self.successor(node)
            node.key = successor.key
            node = successor
        
        # 现在 node 只有一个或没有子节点
        child = node.left if node.left is not None else node.right
        
        if child is not None:
            # Case 2：有一个子节点
            child.parent = node.parent
            if node.parent is None:
                self.root = child
            elif node == node.parent.left:
                node.parent.left = child
            else:
                node.parent.right = child
        else:
            # Case 1：叶子节点
            if node.parent is None:
                self.root = None
            elif node == node.parent.left:
                node.parent.left = None
            else:
                node.parent.right = None
```

```cpp
struct TreeNode {

private:

};

TreeNode* BinarySearchTree::minimumNode(TreeNode* node) {
    while (node->left != nullptr) {
        node = node->left;
    }
    return node;
}

TreeNode* BinarySearchTree::maximumNode(TreeNode* node) {
    while (node->right != nullptr) {
        node = node->right;
    }
    return node;
}

TreeNode* BinarySearchTree::search(int key) {
    TreeNode* current = root;
    while (current != nullptr) {
        if (key == current->key) {
            return current;
        } else if (key < current->key) {
            current = current->left;
        } else {
            current = current->right;
        }
    }
    return nullptr;
}

TreeNode* BinarySearchTree::successor(TreeNode* node) {
    if (node->right != nullptr) {
        return minimumNode(node->right);
    }
    
    TreeNode* parent = node->parent;
    while (parent != nullptr && node == parent->right) {
        node = parent;
        parent = parent->parent;
    }
    return parent;
}

TreeNode* BinarySearchTree::predecessor(TreeNode* node) {
    if (node->left != nullptr) {
        return maximumNode(node->left);
    }
    
    TreeNode* parent = node->parent;
    while (parent != nullptr && node == parent->left) {
        node = parent;
        parent = parent->parent;
    }
    return parent;
}

void BinarySearchTree::insert(int key) {
    TreeNode* newNode = new TreeNode(key);
    
    if (root == nullptr) {
        root = newNode;
        return;
    }
    
    TreeNode* parent = nullptr;
    TreeNode* current = root;
    
    while (current != nullptr) {
        parent = current;
        if (key < current->key) {
            if (current->left == nullptr) {
                current->left = newNode;
                newNode->parent = parent;
                return;
            }
            current = current->left;
        } else {
            if (current->right == nullptr) {
                current->right = newNode;
                newNode->parent = parent;
                return;
            }
            current = current->right;
        }
    }
}

void BinarySearchTree::remove(int key) {
    TreeNode* node = search(key);
    if (node != nullptr) {
        deleteNode(node);
    }
}

void BinarySearchTree::deleteNode(TreeNode* node) {
    // Case 3：两个子节点
    if (node->left != nullptr && node->right != nullptr) {
        TreeNode* succ = successor(node);
        node->key = succ->key;
        node = succ;
    }
    
    // Case 1 & 2：只有一个或没有子节点
    TreeNode* child = (node->left != nullptr) ? node->left : node->right;
    
    if (child != nullptr) {
        child->parent = node->parent;
        if (node->parent == nullptr) {
            root = child;
        } else if (node == node->parent->left) {
            node->parent->left = child;
        } else {
            node->parent->right = child;
        }
    } else {
        if (node->parent == nullptr) {
            root = nullptr;
        } else if (node == node->parent->left) {
            node->parent->left = nullptr;
        } else {
            node->parent->right = nullptr;
        }
    }
    
    delete node;
}

void BinarySearchTree::inorderRecursive(TreeNode* node) {
    if (node != nullptr) {
        inorderRecursive(node->left);
        std::cout << node->key << " ";
        inorderRecursive(node->right);
    }
}

}

void BinarySearchTree::destroyTree(TreeNode* node) {
    if (node != nullptr) {
        destroyTree(node->left);
        destroyTree(node->right);
        delete node;
    }
}
```

## ❓ FAQ - 常见疑问

### Q1：为什么删除有两个子节点的节点要用后继，而不是前驱？

**A1**：理论上用前驱也可以，但有细微差别：

```
┌─────────────────────────────────────────────────────────┐
│  后继的特点：                                            │
│  - 位于被删除节点的右子树                               │
│  - 是右子树中的最小值                                   │
│  - 一定没有左子节点（因为如果左子树非空，左边还有更小的）│
│                                                         │
│  前驱的特点：                                            │
│  - 位于被删除节点的左子树                               │
│  - 是左子树中的最大值                                   │
│  - 一定没有右子节点（因为如果右子树非空，右边还有更大的）│
│                                                         │
│  选择后继的原因：                                        │
│  - 与直觉一致：后继 > 被删除节点 > 前驱                  │
│  - 删除后继对树的结构影响更直观                         │
│  - 实现上更容易理解                                     │
│  - 大多数教材和实现都采用后继                          │
└─────────────────────────────────────────────────────────┘
```

### Q2：为什么有序插入会导致树退化成链表？

**A2**：这是一个递归的过程：

```
┌─────────────────────────────────────────────────────────┐
│  有序插入 1, 2, 3, 4, 5：                                │
│                                                         │
│  插入 1：树 = 1                                          │
│                                                         │
│  插入 2：1 < 2，去右边，右边为空 → 1的右子节点 = 2       │
│        树 = 1                                           │
│              \                                          │
│               2                                         │
│                                                         │
│  插入 3：2 < 3，去右边，右边为空 → 2的右子节点 = 3       │
│        树 = 1                                           │
│              \                                          │
│               2                                         │
│                \                                         │
│                 3                                       │
│                                                         │
│  ... 以此类推                                            │
│                                                         │
│  问题本质：每次插入都比所有现有节点大                    │
│  → 每次都插入右子树的最右边                              │
│  → 树向右单向延伸                                       │
│  → 形成链表                                             │
└─────────────────────────────────────────────────────────┘
```

### Q3：平衡二叉树（如 AVL、红黑树）和普通 BST 有什么区别？

**A3**：核心区别在于"平衡性"：

```
┌─────────────────────────────────────────────────────────┐
│  普通 BST：                                              │
│  - 不保证平衡                                            │
│  - 高度取决于插入顺序                                    │
│  - 最坏 O(n)，期望 O(log n)                              │
│                                                         │
│  AVL 树：                                                │
│  - 严格平衡：左右子树高度差 ≤ 1                          │
│  - 查找效率高                                            │
│  - 插入/删除需要更多旋转操作                             │
│  - 适合查找频繁的场景                                    │
│                                                         │
│  红黑树：                                                │
│  - 近似的平衡：最长路径不超过最短路径的 2 倍              │
│  - 插入/删除效率高                                       │
│  - C++ map/set, Java TreeMap 底层使用                    │
│  - 适合插入删除频繁的场景                                │
│                                                         │
│  性能对比：                                              │
│  ┌──────────┬─────────┬─────────┬──────────┐             │
│  │   类型   │  查找   │  插入   │   删除   │             │
│  ├──────────┼─────────┼─────────┼──────────┤             │
│  │  BST    │ O(log n)│ O(log n)│ O(log n) │             │
│  │  AVL    │ O(log n)│ O(log n)│ O(log n) │             │
│  │ 红黑树  │ O(log n)│ O(log n)│ O(log n) │             │
│  └──────────┴─────────┴─────────┴──────────┘             │
│  （普通 BST 最坏情况是 O(n)）                            │
└─────────────────────────────────────────────────────────┘
```

### Q4：中序遍历为什么能得到有序序列？

**A4**：这由 BST 的性质决定：

```
┌─────────────────────────────────────────────────────────┐
│  BST 性质：左子树 < 根节点 < 右子树                      │
│                                                         │
│  中序遍历顺序：左 → 根 → 右                              │
│                                                         │
│  以节点 8 为例：                                         │
│  中序遍历 8 的左子树 → 得到 < 8 的所有元素（有序）        │
│  访问 8                                                  │
│  中序遍历 8 的右子树 → 得到 > 8 的所有元素（有序）        │
│                                                         │
│  合并：<8 的有序序列 + 8 + >8 的有序序列                  │
│       = 整体有序序列！                                   │
│                                                         │
│  数学归纳法证明：                                        │
│  - 基例：叶子节点遍历 = [自己]，有序 ✓                   │
│  - 归纳：假设左子树和右子树遍历有序                      │
│         合并后 = 左序列 + 根 + 右序列 = 有序 ✓           │
└─────────────────────────────────────────────────────────┘
```

### Q5：为什么实际工程中很少直接使用 BST？

**A5**：因为平衡性难以保证：

```
┌─────────────────────────────────────────────────────────┐
│  问题：                                                  │
│  1. BST 的效率依赖插入顺序                               │
│  2. 程序员无法保证输入是随机的                           │
│  3. 有序输入会导致树退化成链表，效率暴跌                 │
│                                                         │
│  解决方案：                                              │
│  - 使用平衡二叉树（AVL、红黑树）                         │
│  - 由库/框架保证平衡性                                   │
│                                                         │
│  实际应用：                                              │
│  ┌─────────────────────────────────────────────────┐    │
│  │ C++ std::map/std::set      → 红黑树实现          │    │
│  │ Java TreeMap/TreeSet       → 红黑树实现          │    │
│  │ Python sortedcontainers    → 跳表实现            │    │
│  │ Linux 内核红黑树            → 内存管理、调度器     │    │
│  └─────────────────────────────────────────────────┘    │
│                                                         │
│  BST 的意义：                                           │
│  - 理解平衡树的基础                                      │
│  - 算法面试的经典题目                                    │
│  - 某些特殊场景可以直接使用                              │
└─────────────────────────────────────────────────────────┘
```
